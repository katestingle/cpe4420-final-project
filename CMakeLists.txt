cmake_minimum_required(VERSION 3.10)
project(OpenCV_Project)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Locate OpenCV package on the system
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)

# Include OpenCV headers
include_directories(${OpenCV_INCLUDE_DIRS})

# Define executable and its source files
add_executable(baseline baseline.cpp)

# Link OpenCV libraries to executable
target_link_libraries(baseline ${OpenCV_LIBRARIES} Threads::Threads)

# Compile executable with specified optimization options
target_compile_options(baseline PRIVATE -O3 -save-temps -fverbose-asm)

# Clean up intermediate files generated by the compiler due to `-save-temps` option
set_property(DIRECTORY APPEND PROPERTY ADDITIONAL_CLEAN_FILES
    "${CMAKE_CURRENT_BINARY_DIR}/baseline.s"
    "${CMAKE_CURRENT_BINARY_DIR}/baseline.ii"
    "${CMAKE_CURRENT_BINARY_DIR}/baseline.bc"
)
